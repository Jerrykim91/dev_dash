import {resolvePoint} from './world-fixture';
import {test,expect,type Page} from '../frontend/node_modules/@playwright/test';
async function ready(p:Page){await expect(p.locator('.connection')).toContainText('접속');await expect.poll(()=>p.evaluate(()=>!!(window as any).__NOBIS_GAME__?.registry.get('world')?.movement)).toBe(true);}
async function join(p:Page,name:string){await p.goto('/');await p.getByLabel('개발용 닉네임',{exact:true}).fill(name);await p.getByRole('button',{name:'오피스 입장하기'}).click();await ready(p);}
async function own(p:Page){return p.evaluate(()=>{const s=(window as any).__NOBIS_GAME__.registry.get('world');return s.model.players.find((v:any)=>v.id===s.model.session.user.id);});}
async function at(p:Page,x:number,y:number){({x,y}=await resolvePoint(p,x,y,0));await expect.poll(()=>p.evaluate(({x,y})=>{const s=(window as any).__NOBIS_GAME__.registry.get('world');return Math.hypot(s.movement.position.x-x,s.movement.position.y-y);},{x,y}),{timeout:45000}).toBeLessThan(.05);}
async function settle(p:Page){await p.evaluate(()=>new Promise<void>(r=>requestAnimationFrame(()=>requestAnimationFrame(()=>r()))));}
async function screen(p:Page,x:number,y:number,offset=0){({x,y}=await resolvePoint(p,x,y,offset));return p.evaluate(({x,y,offset})=>{const s=(window as any).__NOBIS_GAME__.registry.get('world'),c=s.cameras.main,w=s.coords.toWorld({x,y}),o=c.getWorldPoint(0,0),r=s.game.canvas.getBoundingClientRect();return {x:r.left+(w.x-o.x)*c.zoom,y:r.top+(w.y+offset-o.y)*c.zoom};},{x,y,offset});}
async function clickWorld(p:Page,x:number,y:number,offset=0){await p.getByRole('button',{name:'월드 전체 보기',exact:true}).click();await settle(p);const point=await screen(p,x,y,offset);await p.mouse.click(point.x,point.y);}
async function say(p:Page,text:string){await p.getByLabel('채팅 메시지',{exact:true}).fill(text);await p.getByRole('button',{name:'채팅 보내기'}).click();}
async function saveRefs(p:Page){await p.evaluate(()=>{const w=window as any;w.__originalGame=w.__NOBIS_GAME__;w.__originalScene=w.__NOBIS_GAME__.registry.get('world');w.__originalCanvas=document.querySelector('canvas');});}
async function sameWorld(p:Page){expect(await p.evaluate(()=>{const w=window as any;return w.__originalGame===w.__NOBIS_GAME__&&w.__originalScene===w.__NOBIS_GAME__.registry.get('world')&&w.__originalCanvas===document.querySelector('canvas')&&w.__NOBIS_GAME__.scene.getScenes(true).length===1;})).toBe(true);}

 await a.getByLabel('근무 상태',{exact:true}).selectOption('vacation');await a.getByRole('button',{name:'상태 저장'}).click();await expect.poll(async()=>(await own(a)).work_status).toBe('vacation');expect((await own(a)).zone).toBe('cafe');
 await a.getByLabel('근무 상태',{exact:true}).selectOption('working');await a.getByRole('button',{name:'상태 저장'}).click();await expect.poll(async()=>(await own(a)).participation_mode).toBe('normal');
